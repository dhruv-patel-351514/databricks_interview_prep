CREATE TABLE workspace.healthcare.appointments (
  appointment_id BIGINT,
  patient_id BIGINT,
  provider_id BIGINT,
  appointment_datetime TIMESTAMP,
  appointment_type STRING COLLATE UTF8_BINARY,
  status STRING COLLATE UTF8_BINARY,
  reason_for_visit STRING COLLATE UTF8_BINARY,
  diagnosis_code STRING COLLATE UTF8_BINARY,
  copay_amount DECIMAL(10,2),
  room_number STRING COLLATE UTF8_BINARY,
  notes STRING COLLATE UTF8_BINARY,
  created_at TIMESTAMP);
